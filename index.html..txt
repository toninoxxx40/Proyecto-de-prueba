const questions = [
 {topic:"PRESENT SIMPLE",q:"My brother _____ football every Saturday.",o:["play","plays","played","playing"],a:1,e:"Correct! With he, she or it, we add -s to the verb. ¡Con he, she o it agregamos -s."},
 {topic:"PAST SIMPLE",q:"Yesterday, we _____ a very interesting movie.",o:["watch","watches","watched","watching"],a:2,e:"Correct! Yesterday indicates the past. Watch is a regular verb: watched."},

 {topic:"PAST SIMPLE",q:"They _____ to the beach last weekend.",o:["go","goes","went","going"],a:2,e:"Correct! Went is the past form of the irregular verb go."},
 {topic:"PRESENT SIMPLE",q:"_____ your parents work in a school?",o:["Does","Do","Did","Are"],a:1,e:"Correct! Use Do with I, you, we and they to form present simple questions."},
 {topic:"PAST SIMPLE",q:"I _____ my homework two hours ago.",o:["finish","finishes","finished","finishing"],a:2,e:"Correct! Two hours ago indicates the past. Finish becomes finished."},
 {topic:"PRESENT SIMPLE",q:"Water _____ at 100°C.",o:["boil","boils","boiled","boiling"],a:1,e:"Correct! This is a scientific fact. Use Present Simple."},
 {topic:"PAST SIMPLE",q:"_____ you see the new student yesterday?",o:["Do","Does","Did","Are"],a:2,e:"Correct! Use Did to form questions in the Past Simple."},
 {topic:"PRESENT SIMPLE",q:"My friends usually _____ early.",o:["arrives","arrive","arrived","arriving"],a:1,e:"Correct! With they, use the base form: arrive."},
 {topic:"PAST SIMPLE",q:"She didn't _____ breakfast this morning.",o:["ate","eats","eat","eating"],a:2,e:"Correct! After didn't, always use the base form of the verb: eat."},
 {topic:"PRESENT SIMPLE",q:"Look at the sentence: 'He study English every day.' What is the mistake?",o:["He should be They","Study should be studies","Every day should be yesterday","There is no mistake"],a:1,e:"Correct! Study changes to studies with he: consonant + y becomes -ies."},
 {topic:"PAST SIMPLE",q:"Choose the correct sentence.",o:["We goed to school.","We go to school yesterday.","We went to school yesterday.","We goes to school yesterday."],a:2,e:"Correct! Went is the correct past form of go."},
 {topic:"PRESENT SIMPLE",q:"How often _____ Tom play tennis?",o:["do","does","did","is"],a:1,e:"Correct! Use Does with he, she or it in Present Simple questions."},
 {topic:"PAST SIMPLE",q:"Last night, my sister _____ a book and _____ to bed.",o:["read / went","reads / goes","read / go","reading / went"],a:0,e:"Correct! Read (past, pronounced /red/) and went are past forms."},
 {topic:"MIXED TENSES",q:"My dad _____ coffee every morning, but yesterday he _____ tea.",o:["drinks / drank","drank / drinks","drink / drink","drinks / drink"],a:0,e:"Correct! Every morning = Present Simple; yesterday = Past Simple."}
];

let current=0,score=0,answered=false,order=[];

function startQuiz(){
 document.getElementById("welcome").style.display="none";
 document.getElementById("result").style.display="none";
 document.getElementById("quiz").style.display="block";
 current=0;score=0;answered=false;
 order=questions.map((_,i)=>i);
 order.sort(()=>Math.random()-0.5);
 showQuestion();
}

function showQuestion(){
 answered=false;
 const q=questions[order[current]];
 document.getElementById("counter").textContent=`Question ${current+1}/15`;
 document.getElementById("points").textContent=`⭐ ${score} points`;
 document.getElementById("bar").style.width=`${current/15*100}%`;
 document.getElementById("topic").textContent=q.topic;
 document.getElementById("question").textContent=q.q;
 document.getElementById("feedback").className="feedback";
 document.getElementById("feedback").style.display="none";
 document.getElementById("next").disabled=true;
 document.getElementById("next").textContent=current===14?"SEE RESULTS 🏆":"Next question →";
 const box=document.getElementById("options");
 box.innerHTML="";
 q.o.forEach((option,i)=>{
  const b=document.createElement("button");
  b.className="option";
  b.textContent=String.fromCharCode(65+i)+". "+option;
  b.onclick=()=>checkAnswer(i,b);
  box.appendChild(b);
 });
}

function checkAnswer(choice,button){
 if(answered)return;
 answered=true;
 const q=questions[order[current]];
 const buttons=document.querySelectorAll(".option");
 buttons.forEach((b,i)=>{
  b.disabled=true;
  if(i===q.a)b.classList.add("correct");
 });
 const fb=document.getElementById("feedback");
 if(choice===q.a){
  score++;
  fb.className="feedback good";
  fb.innerHTML="✅ <b>Excellent! Correct answer!</b><br>"+q.e;
 }else{
  button.classList.add("incorrect");
  fb.className="feedback bad";
  fb.innerHTML="❌ <b>Not quite! / ¡No exactamente!</b><br>Correct answer: <b>"+q.o[q.a]+"</b><br>"+q.e;
 }
 fb.style.display="block";
 document.getElementById("points").textContent=`⭐ ${score} points`;
 document.getElementById("bar").style.width=`${(current+1)/15*100}%`;
 document.getElementById("next").disabled=false;
}

function nextQuestion(){
 if(!answered)return;
 current++;
 if(current<questions.length)showQuestion();
 else showResults();
}

function showResults(){
 document.getElementById("quiz").style.display="none";
 document.getElementById("result").style.display="block";
 const name=document.getElementById("student").value.trim();
 document.getElementById("greeting").textContent=name?`Well done, ${name}!`:"Well done!";
 document.getElementById("finalScore").textContent=`${score}/15`;
 const percent=Math.round(score/15*100);
 let message,summary;
 if(score>=13){message="🌟 Excellent work!";summary="You have an excellent understanding of both tenses. ¡Excelente dominio de los tiempos verbales!"}
 else if(score>=10){message="👏 Great job!";summary="You are doing very well. Keep practising irregular verbs and questions."}
 else if(score>=7){message="💪 Good effort!";summary="You are making progress. Review the rules and try again."}
 else{message="📚 Keep practising!";summary="Review the Present Simple and Past Simple rules. You can do it!"}
 document.getElementById("message").textContent=message;
 document.getElementById("summary").textContent=`${percent}% correct · ${summary}`;
 document.getElementById("bar").style.width="100%";
}

function restart(){
 document.getElementById("result").style.display="none";
 document.getElementById("welcome").style.display="block";
 document.getElementById("student").value="";
}
